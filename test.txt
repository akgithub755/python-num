public async Task<(bool Success, string Message)> UpdateRoleAsync(
    uint id, OnDemandRoleDto roleDto, CancellationToken cancellationToken)
{
    if (roleDto == null || string.IsNullOrWhiteSpace(roleDto.RoleName))
        return (false, "Role name is required.");

    var role = await _dbStagingContext.Ondemandroles.FindAsync(new object[] { id }, cancellationToken);

    if (role == null)
        return (false, "Role not found.");

    var name = roleDto.RoleName.Trim();

    // same duplicate rule as CreateRoleAsync, ignoring the role being edited
    var exists = await _dbStagingContext.Ondemandroles
        .AnyAsync(r => r.RoleName == name && r.RoleId != id, cancellationToken);

    if (exists)
        return (false, $"Role '{name}' already exists.");

    role.RoleName = name;
    role.Description = roleDto.Description?.Trim();

    await _dbStagingContext.SaveChangesAsync(cancellationToken);

    _logger.LogInformation("Role updated. RoleId: {RoleId}", id);
    return (true, "Role updated successfully.");
}




[HttpPut("UpdateRole/{id}")]
public async Task<IActionResult> Update(uint id, [FromBody] OnDemandRoleDto role, CancellationToken cancellationToken)
{
    try
    {
        var result = await _roleService.UpdateRoleAsync(id, role, cancellationToken);

        if (!result.Success)
            return BadRequest(result.Message);

        return Ok(result.Message);
    }
    catch (Exception ex)
    {
        _logger.LogError(ex, "Unexpected error while updating role {RoleId}", id);
        return StatusCode(500, "An unexpected error occurred.");
    }
}

private bool CanEdit(OnDemandRoleDto row, Col col) => col.Set is not null;

private bool IsModified(OnDemandRoleDto row) =>
    !newRows.Contains(row) &&
    original.TryGetValue(row, out var json) &&
    json != JsonSerializer.Serialize(row);

private string GetRowClass(OnDemandRoleDto row)
{
    if (row == selectedRow) return "bg-blue-100";
    if (newRows.Contains(row)) return "bg-green-50";
    if (IsModified(row)) return "bg-yellow-50";
    return string.Empty;
}

private void BeginEdit(OnDemandRoleDto row, Col col)
{
    selectedRow = row;

    if (!CanEdit(row, col))
        return;

    editingRow = row;
    editingCol = col.Title;
}

private void EndEdit()
{
    editingRow = null;
    editingCol = null;
}

private string? FindDuplicate(List<OnDemandRoleDto> changed)
{
    foreach (var r in changed)
    {
        var name = (r.RoleName ?? "").Trim();
        var count = rows.Count(x => string.Equals(
            (x.RoleName ?? "").Trim(), name, StringComparison.OrdinalIgnoreCase));

        if (count > 1)
            return $"The role name '{name}' is already present in the database.";
    }

    return null;
}




private async Task Save()
{
    var creates = rows.Where(newRows.Contains).ToList();
    var updates = rows.Where(IsModified).ToList();

    if (creates.Count + updates.Count == 0)
    {
        Toasts.Warning("Please make changes before saving.");
        return;
    }

    if (creates.Concat(updates).Any(r => string.IsNullOrWhiteSpace(r.RoleName)))
    {
        Toasts.Warning("Role name is required.");
        return;
    }

    var duplicate = FindDuplicate(creates.Concat(updates).ToList());
    if (duplicate is not null)
    {
        dialogMessage = duplicate;
        return;
    }

    isSaving = true;

    try
    {
        foreach (var row in creates)
        {
            using var r = await Http.PostAsJsonAsync(CreateUrl, row);

            if (!r.IsSuccessStatusCode)
            {
                Toasts.Error(await ReadMessage(r));
                return;
            }
        }

        foreach (var row in updates)
        {
            using var r = await Http.PutAsJsonAsync($"{UpdateUrl}/{row.RoleId}", row);

            if (!r.IsSuccessStatusCode)
            {
                Toasts.Error(await ReadMessage(r));
                return;
            }
        }

        Toasts.Success("Save successful");
        await Load();
    }
    catch (HttpRequestException ex)
    {
        Toasts.Error($"Unable to save Roles: {ex.Message}");
    }
    catch (Exception ex)
    {
        Toasts.Error($"An unexpected error occurred: {ex.Message}");
    }
    finally
    {
        isSaving = false;
    }
}
