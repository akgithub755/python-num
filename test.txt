var entity = await _context.InputCodesToDownloads
    .FirstOrDefaultAsync(x => x.Id == id, cancellationToken);

if (entity is null)
{
    _logger.LogInformation("InputCodesToDownload not found for delete: {Id}", id);
    return false;
}

_context.InputCodesToDownloads.Remove(entity);
await _context.SaveChangesAsync(cancellationToken);

_logger.LogInformation("InputCodesToDownload deleted: {Id}", id);
return true;
