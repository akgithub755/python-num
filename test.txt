@namespace OnDemandLaunchApp.Pages.Table_Management.Tables
@using System.Net.Http.Json
@using Infrastructure.Models.Responses
@using OnDemandLaunchApp.Services
@inject ToastService Toasts
@* Remove the next two lines if _Imports.razor already injects them *@
@inject HttpClient Http
@inject IJSRuntime JS

<div class="border rounded shadow-sm overflow-hidden flex flex-col">

    @if (isLoading)
    {
        <div class="p-4 text-sm">Loading purchase attributes...</div>
    }
    else if (!string.IsNullOrWhiteSpace(errorMessage))
    {
        <div class="p-4 text-sm alert">@errorMessage</div>
    }
    else
    {
        <div class="overflow-x-auto flex-1">
            <table class="w-full text-sm border-collapse">
                <thead>
                    <tr class="bg-gray-50 border-b">
                        <th class="text-left px-3 py-2 font-medium border-r">idpurchaseattribute (PK)</th>
                        <th class="text-left px-3 py-2 font-medium">description</th>
                    </tr>
                </thead>
                <tbody>
                    @foreach (var row in rows)
                    {
                        <tr class="border-b hover:bg-gray-50 @GetRowClass(row)" @onclick="() => selectedRow = row">
                            <td class="px-3 py-2 border-r text-gray-500">
                                @(newRows.Contains(row) ? "(new)" : row.Idpurchaseattribute.ToString())
                            </td>
                            <td class="px-3 py-2">
                                @if (ReferenceEquals(editingRow, row))
                                {
                                    <input class="w-full border px-1"
                                           @bind="row.Description"
                                           @bind:event="oninput"
                                           @onblur="EndEdit"
                                           @onclick:stopPropagation="true" />
                                }
                                else
                                {
                                    <div class="h-6 cursor-text" @onclick="() => BeginEdit(row)">
                                        @DisplayValue(row.Description)
                                    </div>
                                }
                            </td>
                        </tr>
                    }
                    @if (rows.Count == 0)
                    {
                        <tr><td class="px-3 py-3 text-gray-400" colspan="2">No data found.</td></tr>
                    }
                </tbody>
            </table>
        </div>
    }

    <div class="flex gap-2 justify-start items-center p-3 border-t bg-gray-50">
        <button class="btn-primary text-sm px-3 py-2" @onclick="Refresh" disabled="@(isLoading || isSaving)">Refresh</button>
        <button class="btn-primary text-sm px-3 py-2" @onclick="AddRow" disabled="@(isLoading || isSaving)">Add row</button>
        <button class="btn-primary text-sm px-3 py-2" @onclick="Save" disabled="@(isLoading || isSaving)">@(isSaving ? "Saving..." : "Save")</button>

        <span class="text-xs text-gray-500 ml-2">@rows.Count row(s)</span>
    </div>
</div>

@code {
    private const string Api = "/api/Purchaseattributes";
    private const int MaxPageSize = 1000;

    private List<PurchaseattributeDto> rows = new();
    private Dictionary<long, string?> original = new();
    private readonly HashSet<PurchaseattributeDto> newRows = new();

    private PurchaseattributeDto? selectedRow;
    private PurchaseattributeDto? editingRow;

    private bool isLoading;
    private bool isSaving;
    private string? errorMessage;

    protected override async Task OnInitializedAsync() => await Load();

    private static string DisplayValue(string? v) => string.IsNullOrWhiteSpace(v) ? "--" : v;

    private string GetRowClass(PurchaseattributeDto row)
    {
        if (row == selectedRow) return "bg-blue-100";
        if (newRows.Contains(row)) return "bg-green-50";
        if (IsModified(row)) return "bg-yellow-50";
        return string.Empty;
    }

    private bool IsModified(PurchaseattributeDto row) =>
        !newRows.Contains(row) &&
        original.TryGetValue(row.Idpurchaseattribute, out var d) &&
        d != row.Description;

    private void BeginEdit(PurchaseattributeDto row)
    {
        selectedRow = row;
        editingRow = row;
    }

    private void EndEdit() => editingRow = null;

    private async Task Load()
    {
        isLoading = true;
        errorMessage = null;

        try
        {
            rows = await Http.GetFromJsonAsync<List<PurchaseattributeDto>>(
                       $"{Api}?page=1&pageSize={MaxPageSize}") ?? new();

            original = rows.ToDictionary(r => r.Idpurchaseattribute, r => r.Description);
            newRows.Clear();
            selectedRow = null;
            editingRow = null;
        }
        catch (HttpRequestException ex)
        {
            errorMessage = $"Unable to load Purchase Attribute data: {ex.Message}";
        }
        catch (Exception ex)
        {
            errorMessage = $"An unexpected error occurred: {ex.Message}";
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task Refresh()
    {
        if ((newRows.Count > 0 || rows.Any(IsModified)) &&
            !await JS.InvokeAsync<bool>("confirm", "Discard unsaved changes?"))
            return;

        await Load();

        if (errorMessage is null)
            Toasts.Success("Purchase Attributes refreshed successfully.");
    }

    private void AddRow()
    {
        var row = new PurchaseattributeDto();
        newRows.Add(row);
        rows.Insert(0, row);
        BeginEdit(row);
    }

    private async Task Save()
    {
        var creates = rows.Where(newRows.Contains).ToList();
        var updates = rows.Where(IsModified).ToList();

        if (creates.Count + updates.Count == 0)
        {
            Toasts.Warning("Please make changes before saving.");
            return;
        }

        isSaving = true;

        try
        {
            foreach (var row in creates)
            {
                using var r = await Http.PostAsJsonAsync(Api, row);
                if (!await Check(r, "Unable to create purchase attribute.")) return;
            }

            foreach (var row in updates)
            {
                using var r = await Http.PutAsJsonAsync($"{Api}/{row.Idpurchaseattribute}", row);
                if (!await Check(r, "Unable to save data.")) return;
            }

            Toasts.Success("Save successful");
            await Load();
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to save Purchase Attribute: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task DeleteRow()
    {
        if (selectedRow is null)
            return;

        if (newRows.Contains(selectedRow))
        {
            newRows.Remove(selectedRow);
            rows.Remove(selectedRow);
            selectedRow = null;
            return;
        }

        if (!await JS.InvokeAsync<bool>("confirm", "Delete the selected row?"))
            return;

        isSaving = true;

        try
        {
            using var r = await Http.DeleteAsync($"{Api}/{selectedRow.Idpurchaseattribute}");

            if (await Check(r, "Unable to delete data."))
            {
                Toasts.Success("Row deleted");
                await Load();
            }
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to delete row: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task<bool> Check(HttpResponseMessage response, string fallback)
    {
        if (response.IsSuccessStatusCode)
            return true;

        var message = await response.Content.ReadAsStringAsync();

        Toasts.Error(string.IsNullOrWhiteSpace(message)
            ? $"{fallback} HTTP {(int)response.StatusCode}"
            : message);

        return false;
    }
}
