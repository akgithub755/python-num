    // ---------- Check / uncheck all ----------
    private bool AllSourcesChecked
    {
        get => Sources.Count > 0 && Sources.All(x => x.Checked);
        set => Sources.ForEach(x => x.Checked = value);
    }

    private bool AllNgTypesChecked
    {
        get => NgTypes.Count > 0 && NgTypes.All(x => x.Checked);
        set => NgTypes.ForEach(x => x.Checked = value);
    }

    private bool AllValuesChecked
    {
        get => DisplayedValues.Count > 0 && DisplayedValues.All(x => x.Checked);
        set => DisplayedValues.ForEach(x => x.Checked = value);
    }

    private bool AllShownShopsChecked
    {
        get => DisplayedShops.Count > 0 && DisplayedShops.All(x => x.Checked);
        set => DisplayedShops.ForEach(x => x.Checked = value);
    }

    // ---------- Derived lists ----------
    private List<ValueItem> DisplayedValues
    {
        get
        {
            if (selectedRetailer is null || !ValuesByRetailer.TryGetValue(selectedRetailer, out var list))
                return new();
            return checkedValuesOnly ? list.Where(v => v.Checked).ToList() : list;
        }
    }

    private List<ShopItem> DisplayedShops =>
        Shops.Where(s => (!checkedShopsOnly || s.Checked) &&
                         (string.IsNullOrWhiteSpace(ShopSearch) ||
                          s.Name.Contains(ShopSearch, StringComparison.OrdinalIgnoreCase) ||
                          s.Code.ToString().Contains(ShopSearch)))
             .ToList();

    private int CheckedShopCount => Shops.Count(s => s.Checked);

    // ---------- Actions ----------
    private void SelectRetailer(NamedItem r)
    {
        selectedRetailer = r.Name;
        _ = valuesPagination.SetCurrentPageIndexAsync(0);
    }

    private void ShowAllShops()
    {
        ShopSearch = "";
        checkedShopsOnly = false;
    }

    private void NewParameterSet()
    {
        parameterSetName = "";
        historyPeriods = null;
        Sources.ForEach(x => x.Checked = true);
        NgTypes.ForEach(x => x.Checked = false);
        Retailers.ForEach(x => x.Checked = false);
        Shops.ForEach(x => x.Checked = false);
        foreach (var list in ValuesByRetailer.Values) list.ForEach(v => v.Checked = false);
        selectedRetailer = null;
    }

    public class ShopItem

        public ShopItem(int code, string name) { Code = code; Name = name; }
        public int Code { get; set; }
        public string Name { get; set; }
