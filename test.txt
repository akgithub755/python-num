private string? FindDuplicate(List<DicoviewQuestionDto> creates)
{
    var seen = new HashSet<string>(
        rows.Where(x => !newRows.Contains(x)).Select(x => x.QuestionCode.Trim()),
        StringComparer.OrdinalIgnoreCase);

    foreach (var r in creates)
    {
        var code = r.QuestionCode.Trim();
        if (!seen.Add(code))
            return $"The question code '{code}' is already present in the database.";
    }

    return null;
}



private string? FindDuplicate(List<DicoviewReponseDto> creates)
{
    var seen = rows.Where(x => !newRows.Contains(x))
                   .Select(x => (x.QuestionCode.Trim().ToLowerInvariant(), x.ReponseCode))
                   .ToHashSet();

    foreach (var r in creates)
    {
        var code = r.QuestionCode.Trim();
        if (!seen.Add((code.ToLowerInvariant(), r.ReponseCode)))
            return $"The question code '{code}' with reponse code {r.ReponseCode} is already present in the database.";
    }

    return null;
}



private string? FindDuplicate(List<InputCodesToDownloadDto> creates)
{
    var seen = rows.Where(x => !newRows.Contains(x)).Select(x => x.Id).ToHashSet();

    foreach (var r in creates.Where(r => r.Id != 0))
    {
        if (!seen.Add(r.Id))
            return $"The id {r.Id} is already present in the database.";
    }

    return null;
}



private string? FindDuplicate(List<PurchaseattributeDto> creates)
{

    foreach (var r in creates.Where(r => r.Idpurchaseattribute != 0))
    {
        if (!seen.Add(r.Idpurchaseattribute))

    }

    return null;
}
