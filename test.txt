@using System.Net.Http.Json
@using System.Text.Json
@using Infrastructure.Models.Responses
@using OnDemandLaunchApp.Services
@inject ToastService Toasts
@* Remove the next two lines if _Imports.razor already injects them *@
@inject HttpClient Http
@inject IJSRuntime JS

<div class="border rounded shadow-sm overflow-hidden flex flex-col">

    @if (isLoading)
    {
        <div class="p-4 text-sm">Loading scheduler programs...</div>
    }
    else if (!string.IsNullOrWhiteSpace(errorMessage))
    {
        <div class="p-4 text-sm alert">@errorMessage</div>
    }
    else
    {
        <div style="max-height: 65vh; overflow: auto;">
            <table class="w-full text-sm border-collapse">
                <thead>
                    <tr class="bg-gray-50 border-b">
                        @foreach (var col in cols)
                        {
                            <th class="text-left px-3 py-2 font-medium border-r last:border-r-0"
                                style="position: sticky; top: 0; background: #f9fafb;">@col.Title</th>
                        }
                    </tr>
                </thead>
                <tbody>
                    @foreach (var row in rows)
                    {
                        <tr class="border-b hover:bg-gray-50 @GetRowClass(row)" @onclick="() => selectedRow = row">
                            @foreach (var col in cols)
                            {
                                <td class="px-3 py-2 border-r last:border-r-0">
                                    @if (CanEdit(row, col) &&
                                         (newRows.Contains(row) ||
                                          (ReferenceEquals(editingRow, row) && editingCol == col.Title)))
                                    {
                                        <input class="w-full border px-1"
                                               value="@col.Get(row)"
                                               @oninput="e => Change(row, col, e.Value?.ToString())"
                                               @onblur="EndEdit"
                                               @onclick:stopPropagation="true" />
                                    }
                                    else
                                    {
                                        <div class="h-6 @(CanEdit(row, col) ? "cursor-text" : "text-gray-500")"
                                             @onclick="() => BeginEdit(row, col)">
                                            @DisplayValue(col.Get(row))
                                        </div>
                                    }
                                </td>
                            }
                        </tr>
                    }
                    @if (rows.Count == 0)
                    {
                        <tr><td class="px-3 py-3 text-gray-400" colspan="@cols.Count">No data found.</td></tr>
                    }
                </tbody>
            </table>
        </div>
    }

    <div class="flex gap-2 justify-start items-center p-3 border-t bg-gray-50">
        <button class="btn-primary text-sm px-3 py-2" @onclick="Refresh" disabled="@(isLoading || isSaving)">Refresh</button>
        @if (newRows.Count == 0)
        {
            <button class="btn-primary text-sm px-3 py-2" @onclick="AddRow" disabled="@(isLoading || isSaving)">Add row</button>
        }
        else
        {
            <button class="btn-primary text-sm px-3 py-2" @onclick="CancelAdd" disabled="@(isLoading || isSaving)">Cancel</button>
        }
        <button class="btn-primary text-sm px-3 py-2" @onclick="Save" disabled="@(isLoading || isSaving)">@(isSaving ? "Saving..." : "Save")</button>
        <button class="btn-primary text-sm px-3 py-2" @onclick="DeleteRow" disabled="@(isLoading || isSaving || selectedRow is null)">Delete</button>
        <span class="text-xs text-gray-500 ml-2">@rows.Count row(s)</span>
    </div>
</div>

<MessageDialog Message="@dialogMessage" OnClose="() => dialogMessage = null" />

<MessageDialog Title="Confirm delete"
               Message="@confirmDeleteMessage"
               ShowCancel="true"
               OkText="Delete"
               OnConfirm="ConfirmDelete"
               OnClose="() => confirmDeleteMessage = null" />

<MessageDialog Title="Delete"
               Message="@deleteResultMessage"
               OnClose="() => deleteResultMessage = null" />

<MessageDialog Title="Discard changes"
               Message="@confirmDiscardMessage"
               ShowCancel="true"
               OkText="Discard"
               OnConfirm="ConfirmDiscard"
               OnClose="() => confirmDiscardMessage = null" />

@code {
    private const string Api = "/api/Schedulerprograms";
    private const string ListUrl = "/api/Schedulerprograms/GetShedulerPrograms";   // returns every row, no paging

    private record Col(
        string Title,
        Func<ShedulerProgramsResponses, string?> Get,
        Action<ShedulerProgramsResponses, string?> Set,
        bool NewOnly = false);

    // NewOnly -> editable only on newly added rows.
    // Programcode is the primary key, so it is locked on saved rows.
    private readonly List<Col> cols = new()
    {
        new("Programcode", x => x.Programcode,
            (x, v) => x.Programcode = v?.Trim() ?? string.Empty, true),
        new("Programname", x => x.Programname,
            (x, v) => x.Programname = v ?? string.Empty),
        new("Programdescription", x => x.Programdescription,
            (x, v) => x.Programdescription = v ?? string.Empty),
        new("Pythonversion", x => x.Pythonversion,
            (x, v) => x.Pythonversion = v ?? string.Empty),
    };

    private List<ShedulerProgramsResponses> rows = new();
    private Dictionary<ShedulerProgramsResponses, string> original = new();
    private readonly HashSet<ShedulerProgramsResponses> newRows = new();

    private ShedulerProgramsResponses? selectedRow;
    private ShedulerProgramsResponses? editingRow;
    private string? editingCol;

    private bool isLoading;
    private bool isSaving;
    private string? errorMessage;

    private string? dialogMessage;
    private string? confirmDeleteMessage;
    private string? deleteResultMessage;
    private string? confirmDiscardMessage;

    protected override async Task OnInitializedAsync() => await Load();

    // ---------- helpers ----------

    private static string DisplayValue(string? v) => string.IsNullOrWhiteSpace(v) ? "--" : v;

    private bool CanEdit(ShedulerProgramsResponses row, Col col) =>
        !col.NewOnly || newRows.Contains(row);

    private bool IsModified(ShedulerProgramsResponses row) =>
        !newRows.Contains(row) &&
        original.TryGetValue(row, out var json) &&
        json != JsonSerializer.Serialize(row);

    private string GetRowClass(ShedulerProgramsResponses row)
    {
        if (row == selectedRow) return "bg-blue-100";
        if (newRows.Contains(row)) return "bg-green-50";
        if (IsModified(row)) return "bg-yellow-50";
        return string.Empty;
    }

    private void BeginEdit(ShedulerProgramsResponses row, Col col)
    {
        selectedRow = row;

        if (!CanEdit(row, col))
            return;

        editingRow = row;
        editingCol = col.Title;
    }

    private void EndEdit()
    {
        editingRow = null;
        editingCol = null;
    }

    private void Change(ShedulerProgramsResponses row, Col col, string? value) =>
        col.Set(row, string.IsNullOrWhiteSpace(value) ? null : value);

    private static string KeyUrl(ShedulerProgramsResponses row) =>
        $"{Api}/{Uri.EscapeDataString(row.Programcode)}";

    // shows only the message text, never the raw JSON
    private static async Task<string> ReadMessage(HttpResponseMessage r)
    {
        var body = await r.Content.ReadAsStringAsync();

        try
        {
            using var doc = JsonDocument.Parse(body);

            if (doc.RootElement.ValueKind == JsonValueKind.String)
                return doc.RootElement.GetString() ?? body;

            if (doc.RootElement.ValueKind == JsonValueKind.Object)
            {
                foreach (var p in doc.RootElement.EnumerateObject())
                    if (p.Name.Equals("message", StringComparison.OrdinalIgnoreCase) &&
                        p.Value.ValueKind == JsonValueKind.String)
                        return p.Value.GetString() ?? body;
            }
        }
        catch (Exception)
        {
            // body was plain text, show it as-is
        }

        return string.IsNullOrWhiteSpace(body)
            ? $"Request failed. HTTP {(int)r.StatusCode}"
            : body;
    }

    // ---------- load / refresh ----------

    private async Task Load()
    {
        isLoading = true;
        errorMessage = null;

        try
        {
            rows = await Http.GetFromJsonAsync<List<ShedulerProgramsResponses>>(ListUrl) ?? new();

            original = rows.ToDictionary(r => r, r => JsonSerializer.Serialize(r));
            newRows.Clear();
            selectedRow = null;
            EndEdit();
        }
        catch (HttpRequestException ex)
        {
            errorMessage = $"Unable to load Scheduler Programs data: {ex.Message}";
        }
        catch (Exception ex)
        {
            errorMessage = $"An unexpected error occurred: {ex.Message}";
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task Refresh()
    {
        if (newRows.Count > 0 || rows.Any(IsModified))
        {
            confirmDiscardMessage = "You have unsaved changes. Do you want to discard them and refresh?";
            return;
        }

        await DoRefresh();
    }

    private async Task ConfirmDiscard()
    {
        confirmDiscardMessage = null;
        await DoRefresh();
    }

    private async Task DoRefresh()
    {
        await Load();

        if (errorMessage is null)
            Toasts.Success("Scheduler Programs refreshed successfully.");
    }

    // ---------- add / cancel / save ----------

    private void AddRow()
    {
        var row = new ShedulerProgramsResponses();
        newRows.Add(row);
        rows.Insert(0, row);
        selectedRow = row;
    }

    private void CancelAdd()
    {
        rows.RemoveAll(newRows.Contains);   // drops the unsaved new row(s) from the grid
        newRows.Clear();
        selectedRow = null;
        EndEdit();
    }

    private string? FindDuplicate(List<ShedulerProgramsResponses> creates)
    {
        var seen = new HashSet<string>(
            rows.Where(x => !newRows.Contains(x)).Select(x => x.Programcode.Trim()),
            StringComparer.OrdinalIgnoreCase);

        foreach (var r in creates)
        {
            var code = r.Programcode.Trim();

            if (!seen.Add(code))
                return $"The program code '{code}' is already present in the database.";
        }

        return null;
    }

    private async Task Save()
    {
        var creates = rows.Where(newRows.Contains).ToList();
        var updates = rows.Where(IsModified).ToList();

        if (creates.Count + updates.Count == 0)
        {
            Toasts.Warning("Please make changes before saving.");
            return;
        }

        if (creates.Any(r => string.IsNullOrWhiteSpace(r.Programcode)))
        {
            Toasts.Warning("Programcode is required for new rows.");
            return;
        }

        var duplicate = FindDuplicate(creates);
        if (duplicate is not null)
        {
            dialogMessage = duplicate;
            return;
        }

        isSaving = true;

        try
        {
            foreach (var row in creates)
            {
                using var r = await Http.PostAsJsonAsync(Api, row);   // POST: new rows
                if (!await Check(r)) return;
            }

            foreach (var row in updates)
            {
                using var r = await Http.PutAsJsonAsync(KeyUrl(row), row);   // PUT: edited rows
                if (!await Check(r)) return;
            }

            Toasts.Success("Save successful");
            await Load();
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to save Scheduler Programs: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    // ---------- delete (dialogs, no toasts) ----------

    private void DeleteRow()
    {
        if (selectedRow is null)
            return;

        // unsaved new row: just drop it, nothing to delete on the server
        if (newRows.Contains(selectedRow))
        {
            newRows.Remove(selectedRow);
            rows.Remove(selectedRow);
            selectedRow = null;
            return;
        }

        confirmDeleteMessage = "Are you sure you want to delete the selected row?";
    }

    private async Task ConfirmDelete()
    {
        confirmDeleteMessage = null;

        if (selectedRow is null)
            return;

        isSaving = true;

        try
        {
            using var r = await Http.DeleteAsync(KeyUrl(selectedRow));

            if (r.IsSuccessStatusCode)
            {
                deleteResultMessage = "The row was deleted successfully.";
                await Load();
            }
            else
            {
                deleteResultMessage = await ReadMessage(r);
            }
        }
        catch (Exception ex)
        {
            deleteResultMessage = $"Unable to delete row: {ex.Message}";
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task<bool> Check(HttpResponseMessage response)
    {
        if (response.IsSuccessStatusCode)
            return true;

        Toasts.Error(await ReadMessage(response));
        return false;
    }
}
