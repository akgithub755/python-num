    private const string Api = "/api/Voleqrulesfixedlooses";
    private const int PageSize = 1000;   // the service caps every request at 1000 rows

    private record Col(
        string Title,
        Func<VoleqrulesfixedlooseDto, string?> Get,
        Action<VoleqrulesfixedlooseDto, string?>? Set = null,
        bool NewOnly = false,
        bool IsNumber = false);

    // Set == null -> read-only column, NewOnly -> editable only on newly added rows.
    // The first three columns together are the key, so they are locked on saved rows.
    private readonly List<Col> cols = new()
    {
        new("MasterDictionaryId", x => x.MasterDictionaryId.ToString(),
            (x, v) => { if (uint.TryParse(v, out var n)) x.MasterDictionaryId = n; }, true, true),
        new("AttributeTypeCode", x => x.AttributeTypeCode.ToString(),
            (x, v) => { if (ulong.TryParse(v, out var n)) x.AttributeTypeCode = n; }, true, true),
        new("AttributeTypeValue", x => x.AttributeTypeValue,
            (x, v) => x.AttributeTypeValue = v?.Trim() ?? string.Empty, true),
        new("AttributeTypeDescription", x => x.AttributeTypeDescription,
            (x, v) => x.AttributeTypeDescription = v),
        new("Formula", x => x.Formula,
            (x, v) => x.Formula = v ?? string.Empty),
        new("VolumeEquivalenceId", x => x.VolumeEquivalenceId?.ToString(),
            (x, v) => x.VolumeEquivalenceId = uint.TryParse(v, out var n) ? n : null, false, true),
    };

    private List<VoleqrulesfixedlooseDto> rows = new();
    private Dictionary<VoleqrulesfixedlooseDto, string> original = new();
    private readonly HashSet<VoleqrulesfixedlooseDto> newRows = new();

    private VoleqrulesfixedlooseDto? selectedRow;
    private VoleqrulesfixedlooseDto? editingRow;
    private string? editingCol;

    private bool isLoading;
    private bool isSaving;
    private string? errorMessage;

    private string? dialogMessage;
    private string? confirmDeleteMessage;
    private string? deleteResultMessage;

    protected override async Task OnInitializedAsync() => await Load();

    // ---------- helpers ----------

    private static string DisplayValue(string? v) => string.IsNullOrWhiteSpace(v) ? "--" : v;

    private bool CanEdit(VoleqrulesfixedlooseDto row, Col col) =>
        col.Set is not null && (!col.NewOnly || newRows.Contains(row));

    private bool IsModified(VoleqrulesfixedlooseDto row) =>
        !newRows.Contains(row) &&
        original.TryGetValue(row, out var json) &&
        json != JsonSerializer.Serialize(row);

    private string GetRowClass(VoleqrulesfixedlooseDto row)
    {
        if (row == selectedRow) return "bg-blue-100";
        if (newRows.Contains(row)) return "bg-green-50";
        if (IsModified(row)) return "bg-yellow-50";
        return string.Empty;
    }

    private void BeginEdit(VoleqrulesfixedlooseDto row, Col col)
    {
        selectedRow = row;

        if (!CanEdit(row, col))
            return;

        editingRow = row;
        editingCol = col.Title;
    }

    private void EndEdit()
    {
        editingRow = null;
        editingCol = null;
    }

    private void Change(VoleqrulesfixedlooseDto row, Col col, string? value) =>
        col.Set!(row, string.IsNullOrWhiteSpace(value) ? null : value);

    // three-part key in the URL: /{masterDictionaryId}/{attributeTypeCode}/{attributeTypeValue}
    private static string KeyUrl(VoleqrulesfixedlooseDto row) =>
        $"{Api}/{row.MasterDictionaryId}/{row.AttributeTypeCode}/{Uri.EscapeDataString(row.AttributeTypeValue)}";

    // shows only the message text, never the raw JSON
    private static async Task<string> ReadMessage(HttpResponseMessage r)
    {
        var body = await r.Content.ReadAsStringAsync();

        try
        {
            using var doc = JsonDocument.Parse(body);

            if (doc.RootElement.ValueKind == JsonValueKind.String)
                return doc.RootElement.GetString() ?? body;

            if (doc.RootElement.ValueKind == JsonValueKind.Object)
            {
                foreach (var p in doc.RootElement.EnumerateObject())
                    if (p.Name.Equals("message", StringComparison.OrdinalIgnoreCase) &&
                        p.Value.ValueKind == JsonValueKind.String)
                        return p.Value.GetString() ?? body;
            }
        }
        catch (Exception)
        {
            // body was plain text, show it as-is
        }

        return string.IsNullOrWhiteSpace(body)
            ? $"Request failed. HTTP {(int)r.StatusCode}"
            : body;
    }

    // ---------- load / refresh ----------

    private async Task Load()
    {
        isLoading = true;
        errorMessage = null;

        try
        {
            // the service returns at most 1000 rows per request, so read page after page
            // until a page comes back short. Rows are ordered by key, so paging is stable.
            var all = new List<VoleqrulesfixedlooseDto>();

            for (var page = 1; page <= 1000; page++)
            {
                var batch = await Http.GetFromJsonAsync<List<VoleqrulesfixedlooseDto>>(
                                $"{Api}?page={page}&pageSize={PageSize}") ?? new();

                all.AddRange(batch);

                if (batch.Count < PageSize)
                    break;
            }

            rows = all;
            original = rows.ToDictionary(r => r, r => JsonSerializer.Serialize(r));
            newRows.Clear();
            selectedRow = null;
            EndEdit();
        }
        catch (HttpRequestException ex)
        {
            errorMessage = $"Unable to load Voleqrulesfixedloose data: {ex.Message}";
        }
        catch (Exception ex)
        {
            errorMessage = $"An unexpected error occurred: {ex.Message}";
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task Refresh()
    {
        if ((newRows.Count > 0 || rows.Any(IsModified)) &&
            !await JS.InvokeAsync<bool>("confirm", "Discard unsaved changes?"))
            return;

        await Load();

        if (errorMessage is null)
            Toasts.Success("Voleqrulesfixedloose refreshed successfully.");
    }

    // ---------- add / save ----------

    private void AddRow()
    {
        var row = new VoleqrulesfixedlooseDto
        {
            AttributeTypeValue = string.Empty,
            Formula = string.Empty
        };

        newRows.Add(row);
        rows.Insert(0, row);
        selectedRow = row;
        editingRow = row;
        editingCol = "MasterDictionaryId";
    }

    // the key is the combination of the three key columns
    private string? FindDuplicate(List<VoleqrulesfixedlooseDto> creates)
    {
        var seen = rows.Where(x => !newRows.Contains(x))
                       .Select(x => (x.MasterDictionaryId, x.AttributeTypeCode,
                                     x.AttributeTypeValue.Trim().ToLowerInvariant()))
                       .ToHashSet();

        foreach (var r in creates)
        {
            var value = r.AttributeTypeValue.Trim();

            if (!seen.Add((r.MasterDictionaryId, r.AttributeTypeCode, value.ToLowerInvariant())))
                return $"The combination MasterDictionaryId {r.MasterDictionaryId}, " +
                       $"AttributeTypeCode {r.AttributeTypeCode} and AttributeTypeValue '{value}' " +
                       "is already present in the database.";
        }

        return null;
    }

    private async Task Save()
    {
        var creates = rows.Where(newRows.Contains).ToList();
        var updates = rows.Where(IsModified).ToList();

        if (creates.Count + updates.Count == 0)
        {
            Toasts.Warning("Please make changes before saving.");
            return;
        }

        if (creates.Any(r => string.IsNullOrWhiteSpace(r.AttributeTypeValue)))
        {
            Toasts.Warning("AttributeTypeValue is required for new rows.");
            return;
        }

        var duplicate = FindDuplicate(creates);
        if (duplicate is not null)
        {
            dialogMessage = duplicate;
            return;
        }

        isSaving = true;

        try
        {
            foreach (var row in creates)
            {
                using var r = await Http.PostAsJsonAsync(Api, row);   // POST: new rows
                if (!await Check(r)) return;
            }

            foreach (var row in updates)
            {
                using var r = await Http.PutAsJsonAsync(KeyUrl(row), row);   // PUT: edited rows
                if (!await Check(r)) return;
            }

            Toasts.Success("Save successful");
            await Load();
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to save Voleqrulesfixedloose: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    // ---------- delete (dialogs, no toasts) ----------

    private void DeleteRow()
    {
        if (selectedRow is null)
            return;

        // unsaved new row: just drop it, nothing to delete on the server
        if (newRows.Contains(selectedRow))
        {
            newRows.Remove(selectedRow);
            rows.Remove(selectedRow);
            selectedRow = null;
            return;
        }

        confirmDeleteMessage = "Are you sure you want to delete the selected row?";
    }

    private async Task ConfirmDelete()
    {
        confirmDeleteMessage = null;

        if (selectedRow is null)
            return;

        isSaving = true;

        try
        {
            using var r = await Http.DeleteAsync(KeyUrl(selectedRow));

            if (r.IsSuccessStatusCode)
            {
                deleteResultMessage = "The row was deleted successfully.";
                await Load();
            }
            else
            {
                deleteResultMessage = await ReadMessage(r);
            }
        }
        catch (Exception ex)
        {
            deleteResultMessage = $"Unable to delete row: {ex.Message}";
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task<bool> Check(HttpResponseMessage response)
    {
        if (response.IsSuccessStatusCode)
            return true;

        Toasts.Error(await ReadMessage(response));
        return false;
    }
