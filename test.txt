@namespace OnDemandLaunchApp.Pages.Table_Management.Tables
@using System.Net.Http.Json
@using System.Text.Json
@using Infrastructure.Models.Responses
@using OnDemandLaunchApp.Services
@inject ToastService Toasts
@* Remove the next two lines if _Imports.razor already injects them *@
@inject HttpClient Http
@inject IJSRuntime JS

<div class="border rounded shadow-sm overflow-hidden flex flex-col">

    @if (isLoading)
    {
        <div class="p-4 text-sm">Loading input codes to download...</div>
    }
    else if (!string.IsNullOrWhiteSpace(errorMessage))
    {
        <div class="p-4 text-sm alert">@errorMessage</div>
    }
    else
    {
        <div class="overflow-x-auto flex-1">
            <table class="w-full text-sm border-collapse">
                <thead>
                    <tr class="bg-gray-50 border-b">
                        @foreach (var col in cols)
                        {
                            <th class="text-left px-3 py-2 font-medium border-r last:border-r-0">@col.Title</th>
                        }
                    </tr>
                </thead>
                <tbody>
                    @foreach (var row in rows)
                    {
                        <tr class="border-b hover:bg-gray-50 @GetRowClass(row)" @onclick="() => selectedRow = row">
                            @foreach (var col in cols)
                            {
                                <td class="px-3 py-2 border-r last:border-r-0">
                                    @if (col.Set is not null &&
                                         ReferenceEquals(editingRow, row) &&
                                         editingCol == col.Title)
                                    {
                                        <input class="w-full border px-1"
                                               type="@(col.IsNumber ? "number" : "text")"
                                               value="@col.Get(row)"
                                               @oninput="e => Change(row, col, e.Value?.ToString())"
                                               @onblur="EndEdit"
                                               @onclick:stopPropagation="true" />
                                    }
                                    else
                                    {
                                        <div class="h-6 @(col.Set is not null ? "cursor-text" : "text-gray-500")"
                                             @onclick="() => BeginEdit(row, col)">
                                            @DisplayValue(col.Get(row))
                                        </div>
                                    }
                                </td>
                            }
                        </tr>
                    }
                    @if (rows.Count == 0)
                    {
                        <tr><td class="px-3 py-3 text-gray-400" colspan="@cols.Count">No data found.</td></tr>
                    }
                </tbody>
            </table>
        </div>
    }

    <div class="flex gap-2 justify-start items-center p-3 border-t bg-gray-50">
        <button class="btn-primary text-sm px-3 py-2" @onclick="Refresh" disabled="@(isLoading || isSaving)">Refresh</button>
        <button class="btn-primary text-sm px-3 py-2" @onclick="AddRow" disabled="@(isLoading || isSaving)">Add row</button>
        <button class="btn-primary text-sm px-3 py-2" @onclick="Save" disabled="@(isLoading || isSaving)">@(isSaving ? "Saving..." : "Save")</button>
        <button class="btn-primary text-sm px-3 py-2" @onclick="DeleteRow" disabled="@(isLoading || isSaving || selectedRow is null)">Delete</button>
        <span class="text-xs text-gray-500 ml-2">@rows.Count row(s)</span>
    </div>
</div>

@code {
    private const string Api = "/api/InputCodesToDownloads";
    private const int MaxPageSize = 1000;

    private record Col(
        string Title,
        Func<InputCodesToDownloadDto, string?> Get,
        Action<InputCodesToDownloadDto, string?>? Set = null,
        bool IsNumber = false);

    // Set == null -> read-only column
    private readonly List<Col> cols = new()
    {
        new("Id", x => x.Id.ToString()),
        new("TypeDesc", x => x.TypeDesc, (x, v) => x.TypeDesc = v ?? string.Empty),
        new("InsertDate", x => x.InsertDate.ToString("yyyy-MM-dd HH:mm")),
        new("Userid", x => x.Userid, (x, v) => x.Userid = v ?? string.Empty),
        new("Commentaire", x => x.Commentaire, (x, v) => x.Commentaire = v),
        new("Pays", x => x.Pays, (x, v) => x.Pays = v ?? string.Empty),
        new("ProcessClassique", x => x.ProcessClassique.ToString(),
            (x, v) => { if (int.TryParse(v, out var n)) x.ProcessClassique = n; }, true),
        new("ProcessValidate", x => x.ProcessValidate.ToString(),
            (x, v) => { if (int.TryParse(v, out var n)) x.ProcessValidate = n; }, true),
    };

    private List<InputCodesToDownloadDto> rows = new();
    private Dictionary<uint, string> original = new();
    private readonly HashSet<InputCodesToDownloadDto> newRows = new();

    private InputCodesToDownloadDto? selectedRow;
    private InputCodesToDownloadDto? editingRow;
    private string? editingCol;

    private bool isLoading;
    private bool isSaving;
    private string? errorMessage;

    protected override async Task OnInitializedAsync() => await Load();

    // ---------- helpers ----------

    private static string DisplayValue(string? v) => string.IsNullOrWhiteSpace(v) ? "--" : v;

    private bool IsModified(InputCodesToDownloadDto row) =>
        !newRows.Contains(row) &&
        original.TryGetValue(row.Id, out var json) &&
        json != JsonSerializer.Serialize(row);

    private string GetRowClass(InputCodesToDownloadDto row)
    {
        if (row == selectedRow) return "bg-blue-100";
        if (newRows.Contains(row)) return "bg-green-50";
        if (IsModified(row)) return "bg-yellow-50";
        return string.Empty;
    }

    private void BeginEdit(InputCodesToDownloadDto row, Col col)
    {
        selectedRow = row;

        if (col.Set is null)
            return;

        editingRow = row;
        editingCol = col.Title;
    }

    private void EndEdit()
    {
        editingRow = null;
        editingCol = null;
    }

    private void Change(InputCodesToDownloadDto row, Col col, string? value) =>
        col.Set!(row, string.IsNullOrWhiteSpace(value) ? null : value);

    // ---------- load / refresh ----------

    private async Task Load()
    {
        isLoading = true;
        errorMessage = null;

        try
        {
            rows = await Http.GetFromJsonAsync<List<InputCodesToDownloadDto>>(
                       $"{Api}?page=1&pageSize={MaxPageSize}") ?? new();

            original = rows.ToDictionary(r => r.Id, r => JsonSerializer.Serialize(r));
            newRows.Clear();
            selectedRow = null;
            EndEdit();
        }
        catch (HttpRequestException ex)
        {
            errorMessage = $"Unable to load Input Codes data: {ex.Message}";
        }
        catch (Exception ex)
        {
            errorMessage = $"An unexpected error occurred: {ex.Message}";
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task Refresh()
    {
        if ((newRows.Count > 0 || rows.Any(IsModified)) &&
            !await JS.InvokeAsync<bool>("confirm", "Discard unsaved changes?"))
            return;

        await Load();

        if (errorMessage is null)
            Toasts.Success("Input Codes refreshed successfully.");
    }

    // ---------- add / save / delete ----------

    private void AddRow()
    {
        var row = new InputCodesToDownloadDto { InsertDate = DateTime.Now };
        newRows.Add(row);
        rows.Insert(0, row);
        selectedRow = row;
        editingRow = row;
        editingCol = "TypeDesc";
    }

    private async Task Save()
    {
        var creates = rows.Where(newRows.Contains).ToList();
        var updates = rows.Where(IsModified).ToList();

        if (creates.Count + updates.Count == 0)
        {
            Toasts.Warning("Please make changes before saving.");
            return;
        }

        isSaving = true;

        try
        {
            foreach (var row in creates)
            {
                using var r = await Http.PostAsJsonAsync(Api, row);
                if (!await Check(r, "Unable to create item.")) return;
            }

            foreach (var row in updates)
            {
                using var r = await Http.PutAsJsonAsync($"{Api}/{row.Id}", row);
                if (!await Check(r, "Unable to save data.")) return;
            }

            Toasts.Success("Save successful");
            await Load();
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to save Input Codes: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task DeleteRow()
    {
        if (selectedRow is null)
            return;

        if (newRows.Contains(selectedRow))
        {
            newRows.Remove(selectedRow);
            rows.Remove(selectedRow);
            selectedRow = null;
            return;
        }

        if (!await JS.InvokeAsync<bool>("confirm", "Delete the selected row?"))
            return;

        isSaving = true;

        try
        {
            using var r = await Http.DeleteAsync($"{Api}/{selectedRow.Id}");

            if (await Check(r, "Unable to delete data."))
            {
                Toasts.Success("Row deleted");
                await Load();
            }
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to delete row: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task<bool> Check(HttpResponseMessage response, string fallback)
    {
        if (response.IsSuccessStatusCode)
            return true;

        var message = await response.Content.ReadAsStringAsync();

        Toasts.Error(string.IsNullOrWhiteSpace(message)
            ? $"{fallback} HTTP {(int)response.StatusCode}"
            : message);

        return false;
    }
}
