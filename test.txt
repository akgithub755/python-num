
import json
import logging

from langchain.tools import tool
from database import get_connection

logging.basicConfig(
    filename="logs/agent.log",
    level=logging.INFO,
    format="%(asctime)s - %(levelname)s - %(message)s"
)


@tool
def get_risk_summary() -> str:
    """Get overall transaction statistics including
    total transactions, fraud count, fraud rate,
    average transaction amount and failed transactions."""

    conn = get_connection()

    try:
        result = conn.execute("""
            SELECT
                COUNT(*) AS total_transactions,
                SUM(fraud_flag) AS fraud_count,
                ROUND(
                    100.0 * SUM(fraud_flag) / COUNT(*), 2
                ) AS fraud_percentage,
                ROUND(AVG(amount), 2) AS avg_amount,
                SUM(
                    CASE WHEN status = 'FAILED'
                    THEN 1 ELSE 0 END
                ) AS failed_transactions
            FROM transactions
        """).fetchdf()

        logging.info("Risk summary tool executed")

        return result.to_json(orient="records")

    finally:
        conn.close()


@tool
def find_high_value_transactions(
    min_amount: float = 100000,
    limit: int = 20
) -> str:
    """Find high-value transactions above a given amount.
    Use this when the user asks about large or expensive
    transactions. Amount is in Indian rupees."""

    min_amount = max(0, min_amount)
    limit = max(1, min(limit, 100))

    conn = get_connection()

    try:
        result = conn.execute("""
            SELECT
                transaction_id,
                customer_id,
                amount,
                city,
                status,
                fraud_flag,
                transaction_date
            FROM transactions
            WHERE amount >= ?
            ORDER BY amount DESC
            LIMIT ?
        """, [min_amount, limit]).fetchdf()

        logging.info(
            "High value transaction search: %s",
            min_amount
        )

        return result.to_json(orient="records")

    finally:
        conn.close()


@tool
def analyze_city_risk() -> str:
    """Analyze fraud counts, transaction volumes,
    fraud percentages and average transaction amounts
    for each city."""

    conn = get_connection()

    try:
        result = conn.execute("""
            SELECT
                city,
                COUNT(*) AS total_transactions,
                SUM(fraud_flag) AS fraud_count,
                ROUND(
                    100.0 * SUM(fraud_flag) / COUNT(*), 2
                ) AS fraud_percentage,
                ROUND(AVG(amount), 2) AS average_amount
            FROM transactions
            GROUP BY city
            ORDER BY fraud_percentage DESC
        """).fetchdf()

        logging.info("City risk analysis executed")

        return result.to_json(orient="records")

    finally:
        conn.close()


@tool
def get_customer_profile(customer_id: int) -> str:
    """Retrieve a customer profile and their transaction
    statistics. Requires a customer ID."""

    if customer_id < 1:
        return "Invalid customer ID."

    conn = get_connection()

    try:
        result = conn.execute("""
            SELECT
                c.customer_id,
                c.customer_name,
                c.age,
                c.city,
                c.account_balance,
                c.risk_score,
                c.account_type,
                COUNT(t.transaction_id) AS total_transactions,
                ROUND(SUM(t.amount), 2) AS total_amount,
                SUM(t.fraud_flag) AS fraud_count
            FROM customers c
            LEFT JOIN transactions t
                ON c.customer_id = t.customer_id
            WHERE c.customer_id = ?
            GROUP BY ALL
        """, [customer_id]).fetchdf()

        if result.empty:
            return "Customer not found."

        logging.info(
            "Customer profile retrieved: %s",
            customer_id
        )

        return result.to_json(orient="records")

    finally:
        conn.close()


@tool
def find_frequent_failures(min_failed: int = 3) -> str:
    """Find customers who have multiple failed transactions.
    Useful for investigating repeated payment failures."""

    min_failed = max(1, min(min_failed, 1000))

    conn = get_connection()

    try:
        result = conn.execute("""
            SELECT
                customer_id,
                COUNT(*) AS failed_count,
                ROUND(SUM(amount), 2) AS failed_amount
            FROM transactions
            WHERE status = 'FAILED'
            GROUP BY customer_id
            HAVING COUNT(*) >= ?
            ORDER BY failed_count DESC
            LIMIT 100
        """, [min_failed]).fetchdf()

        logging.info("Failed transaction analysis executed")

        return result.to_json(orient="records")

    finally:
        conn.close()
