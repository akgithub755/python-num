<MessageDialog Title="Discard changes"
               Message="@confirmDiscardMessage"
               ShowCancel="true"
               OkText="Discard"
               OnConfirm="ConfirmDiscard"
               OnClose="() => confirmDiscardMessage = null" />




private string? confirmDiscardMessage;

private async Task Refresh()
{
    if (newRows.Count > 0 || rows.Any(IsModified))
    {
        confirmDiscardMessage = "You have unsaved changes. Do you want to discard them and refresh?";
        return;
    }

    await DoRefresh();
}

private async Task ConfirmDiscard()
{
    confirmDiscardMessage = null;
    await DoRefresh();
}

private async Task DoRefresh()
{
    await Load();

    if (errorMessage is null)
        Toasts.Success("Purchase Attributes refreshed successfully.");   // keep this file's existing text
}
