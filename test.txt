                                         ReferenceEquals(editingRow, row) &&

                                    {
                                        <input class="w-full border px-1"
                                               value="@col.Get(row)"
                                               @oninput="e => Change(row, col, e.Value?.ToString())"
                                               @onblur="EndEdit"
                                               @onclick:stopPropagation="true" />
                                    }
                                    else
                                    {
                                        <div class="h-6 @(CanEdit(row, col) ? "cursor-text" : "text-gray-500")"
                                             @onclick="() => BeginEdit(row, col)">

                                    }
                                </td>
                            }
                        </tr>
                    }

@code {
    private const string Api = "/api/DicoviewQuestions";
    private const int PageSize = 100000;   // one request, loads the whole table

    private record Col(
        string Title,
        Func<DicoviewQuestionDto, string?> Get,
        Action<DicoviewQuestionDto, string?>? Set = null,
        bool NewOnly = false);

    // Set == null -> read-only column, NewOnly -> editable only on newly added rows
    private readonly List<Col> cols = new()
    {
        new("QuestionCode", x => x.QuestionCode, (x, v) => x.QuestionCode = v?.Trim() ?? string.Empty, true),
        new("QuestionDesc", x => x.QuestionDesc, (x, v) => x.QuestionDesc = v),
        new("TypeQuestion", x => x.TypeQuestion, (x, v) => x.TypeQuestion = v),
        new("InsertDate", x => x.InsertDate),
        new("UpdateDate", x => x.UpdateDate),
    };

    private List<DicoviewQuestionDto> rows = new();
    private Dictionary<DicoviewQuestionDto, string> original = new();
    private readonly HashSet<DicoviewQuestionDto> newRows = new();

    private DicoviewQuestionDto? selectedRow;
    private DicoviewQuestionDto? editingRow;
    private string? editingCol;

    private bool isLoading;
    private bool isSaving;
    private string? errorMessage;

    protected override async Task OnInitializedAsync() => await Load();

    // ---------- helpers ----------

    private static string DisplayValue(string? v) => string.IsNullOrWhiteSpace(v) ? "--" : v;

    private bool CanEdit(DicoviewQuestionDto row, Col col) =>
        col.Set is not null && (!col.NewOnly || newRows.Contains(row));

    private bool IsModified(DicoviewQuestionDto row) =>
        !newRows.Contains(row) &&
        original.TryGetValue(row, out var json) &&
        json != JsonSerializer.Serialize(row);

    private string GetRowClass(DicoviewQuestionDto row)
    {
        if (row == selectedRow) return "bg-blue-100";
        if (newRows.Contains(row)) return "bg-green-50";
        if (IsModified(row)) return "bg-yellow-50";
        return string.Empty;
    }

    private void BeginEdit(DicoviewQuestionDto row, Col col)
    {
        selectedRow = row;

        if (!CanEdit(row, col))
            return;

        editingRow = row;
        editingCol = col.Title;
    }

    private void EndEdit()
    {
        editingRow = null;
        editingCol = null;
    }

    private void Change(DicoviewQuestionDto row, Col col, string? value) =>
        col.Set!(row, string.IsNullOrWhiteSpace(value) ? null : value);

    // ---------- load / refresh ----------

    private async Task Load()
    {
        isLoading = true;
        errorMessage = null;

        try
        {
            rows = await Http.GetFromJsonAsync<List<DicoviewQuestionDto>>(
                       $"{Api}?page=1&pageSize={PageSize}") ?? new();

            original = rows.ToDictionary(r => r, r => JsonSerializer.Serialize(r));
            newRows.Clear();
            selectedRow = null;
            EndEdit();
        }
        catch (HttpRequestException ex)
        {
            errorMessage = $"Unable to load Dicoview Questions data: {ex.Message}";
        }
        catch (Exception ex)
        {
            errorMessage = $"An unexpected error occurred: {ex.Message}";
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task Refresh()
    {
        if ((newRows.Count > 0 || rows.Any(IsModified)) &&
            !await JS.InvokeAsync<bool>("confirm", "Discard unsaved changes?"))
            return;

        await Load();

        if (errorMessage is null)
            Toasts.Success("Dicoview Questions refreshed successfully.");
    }

    // ---------- add / save / delete ----------

    private void AddRow()
    {
        // the DTO's InsertDate / UpdateDate are non-nullable strings, so give them a value
        var now = DateTime.Now.ToString("yyyy-MM-dd HH:mm:ss");

        var row = new DicoviewQuestionDto
        {
            QuestionCode = string.Empty,
            InsertDate = now,
            UpdateDate = now
        };

        newRows.Add(row);
        rows.Insert(0, row);
        selectedRow = row;
        editingRow = row;
        editingCol = "QuestionCode";
    }

    private async Task Save()
    {
        var creates = rows.Where(newRows.Contains).ToList();
        var updates = rows.Where(IsModified).ToList();

        if (creates.Count + updates.Count == 0)
        {
            Toasts.Warning("Please make changes before saving.");
            return;
        }

        if (creates.Any(r => string.IsNullOrWhiteSpace(r.QuestionCode)))
        {
            Toasts.Warning("QuestionCode is required for new rows.");
            return;
        }

        isSaving = true;

        try
        {
            foreach (var row in creates)
            {
                using var r = await Http.PostAsJsonAsync(Api, row);
                if (!await Check(r, "Unable to create question.")) return;
            }

            foreach (var row in updates)
            {
                using var r = await Http.PutAsJsonAsync(
                    $"{Api}/{Uri.EscapeDataString(row.QuestionCode)}", row);
                if (!await Check(r, "Unable to save data.")) return;
            }

            Toasts.Success("Save successful");
            await Load();
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to save Dicoview Questions: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task DeleteRow()
    {
        if (selectedRow is null)
            return;

        if (newRows.Contains(selectedRow))
        {
            newRows.Remove(selectedRow);
            rows.Remove(selectedRow);
            selectedRow = null;
            return;
        }

        if (!await JS.InvokeAsync<bool>("confirm", "Delete the selected row?"))
            return;

        isSaving = true;

        try
        {
            using var r = await Http.DeleteAsync(
                $"{Api}/{Uri.EscapeDataString(selectedRow.QuestionCode)}");

            if (await Check(r, "Unable to delete data."))
            {
                Toasts.Success("Row deleted");
                await Load();
            }
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to delete row: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task<bool> Check(HttpResponseMessage response, string fallback)
    {
        if (response.IsSuccessStatusCode)
            return true;

        var message = await response.Content.ReadAsStringAsync();

        Toasts.Error(string.IsNullOrWhiteSpace(message)
            ? $"{fallback} HTTP {(int)response.StatusCode}"
            : message);

        return false;
    }
}
