@code {
    private const string Api = "/api/RetailerAttributes";
    private const int PageSize = 1000;   // the service caps every request at 1000 rows

    private record Col(
        string Title,
        Func<RetailerAttributeDto, string?> Get,
        Action<RetailerAttributeDto, string?>? Set = null,
        bool NewOnly = false,
        bool IsBool = false);

    // Set == null -> read-only column, NewOnly -> editable only on newly added rows.
    // AttributeCode is the key, so it is locked on saved rows.
    private readonly List<Col> cols = new()
    {
        new("AttributeCode", x => x.AttributeCode,
            (x, v) => x.AttributeCode = v?.Trim() ?? string.Empty, true),
        new("AttributeType", x => x.AttributeType, (x, v) => x.AttributeType = v),
        new("AttributeTypeSyndCode", x => x.AttributeTypeSyndCode, (x, v) => x.AttributeTypeSyndCode = v),
        new("Owner", x => x.Owner, (x, v) => x.Owner = v),
        new("Signoff", x => x.Signoff?.ToString().ToLowerInvariant(),
            (x, v) => x.Signoff = v == "true" ? true : v == "false" ? false : null, false, true),
    };

    private List<RetailerAttributeDto> rows = new();
    private Dictionary<RetailerAttributeDto, string> original = new();
    private readonly HashSet<RetailerAttributeDto> newRows = new();

    private RetailerAttributeDto? selectedRow;
    private RetailerAttributeDto? editingRow;
    private string? editingCol;

    private bool isLoading;
    private bool isSaving;
    private string? errorMessage;

    private string? dialogMessage;
    private string? confirmDeleteMessage;
    private string? deleteResultMessage;
    private string? confirmDiscardMessage;

    protected override async Task OnInitializedAsync() => await Load();

    // ---------- helpers ----------

    private static string DisplayValue(string? v) => string.IsNullOrWhiteSpace(v) ? "--" : v;

    private static string ShowValue(RetailerAttributeDto row, Col col)
    {
        var v = col.Get(row);

        if (col.IsBool)
            return v == "true" ? "Yes" : v == "false" ? "No" : "--";

        return DisplayValue(v);
    }

    private bool CanEdit(RetailerAttributeDto row, Col col) =>
        col.Set is not null && (!col.NewOnly || newRows.Contains(row));

    private bool IsModified(RetailerAttributeDto row) =>
        !newRows.Contains(row) &&
        original.TryGetValue(row, out var json) &&
        json != JsonSerializer.Serialize(row);

    private string GetRowClass(RetailerAttributeDto row)
    {
        if (row == selectedRow) return "bg-blue-100";
        if (newRows.Contains(row)) return "bg-green-50";
        if (IsModified(row)) return "bg-yellow-50";
        return string.Empty;
    }

    private void BeginEdit(RetailerAttributeDto row, Col col)
    {
        selectedRow = row;

        if (!CanEdit(row, col))
            return;

        editingRow = row;
        editingCol = col.Title;
    }

    private void EndEdit()
    {
        editingRow = null;
        editingCol = null;
    }

    private void Change(RetailerAttributeDto row, Col col, string? value) =>
        col.Set!(row, string.IsNullOrWhiteSpace(value) ? null : value);

    private static string KeyUrl(RetailerAttributeDto row) =>
        $"{Api}/{Uri.EscapeDataString(row.AttributeCode)}";

    // shows only the message text, never the raw JSON
    private static async Task<string> ReadMessage(HttpResponseMessage r)
    {
        var body = await r.Content.ReadAsStringAsync();

        try
        {
            using var doc = JsonDocument.Parse(body);

            if (doc.RootElement.ValueKind == JsonValueKind.String)
                return doc.RootElement.GetString() ?? body;

            if (doc.RootElement.ValueKind == JsonValueKind.Object)
            {
                foreach (var p in doc.RootElement.EnumerateObject())
                    if (p.Name.Equals("message", StringComparison.OrdinalIgnoreCase) &&
                        p.Value.ValueKind == JsonValueKind.String)
                        return p.Value.GetString() ?? body;
            }
        }
        catch (Exception)
        {
            // body was plain text, show it as-is
        }

        return string.IsNullOrWhiteSpace(body)
            ? $"Request failed. HTTP {(int)r.StatusCode}"
            : body;
    }

    // ---------- load / refresh ----------

    private async Task Load()
    {
        isLoading = true;
        errorMessage = null;

        try
        {
            // the service returns at most 1000 rows per request, so read page after page
            // until a page comes back short. Rows are ordered by AttributeCode, so paging is stable.
            var all = new List<RetailerAttributeDto>();

            for (var page = 1; page <= 1000; page++)
            {
                var batch = await Http.GetFromJsonAsync<List<RetailerAttributeDto>>(
                                $"{Api}?page={page}&pageSize={PageSize}") ?? new();

                all.AddRange(batch);

                if (batch.Count < PageSize)
                    break;
            }

            rows = all;
            original = rows.ToDictionary(r => r, r => JsonSerializer.Serialize(r));
            newRows.Clear();
            selectedRow = null;
            EndEdit();
        }
        catch (HttpRequestException ex)
        {
            errorMessage = $"Unable to load Retailer Attributes data: {ex.Message}";
        }
        catch (Exception ex)
        {
            errorMessage = $"An unexpected error occurred: {ex.Message}";
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task Refresh()
    {
        if (newRows.Count > 0 || rows.Any(IsModified))
        {
            confirmDiscardMessage = "You have unsaved changes. Do you want to discard them and refresh?";
            return;
        }

        await DoRefresh();
    }

    private async Task ConfirmDiscard()
    {
        confirmDiscardMessage = null;
        await DoRefresh();
    }

    private async Task DoRefresh()
    {
        await Load();

        if (errorMessage is null)
            Toasts.Success("Retailer Attributes refreshed successfully.");
    }

    // ---------- add / cancel / save ----------

    private void AddRow()
    {
        var row = new RetailerAttributeDto { AttributeCode = string.Empty };
        newRows.Add(row);
        rows.Insert(0, row);
        selectedRow = row;
    }

    private void CancelAdd()
    {
        rows.RemoveAll(newRows.Contains);   // drops the unsaved new row(s) from the grid
        newRows.Clear();
        selectedRow = null;
        EndEdit();
    }

    private string? FindDuplicate(List<RetailerAttributeDto> creates)
    {
        var seen = new HashSet<string>(
            rows.Where(x => !newRows.Contains(x)).Select(x => x.AttributeCode.Trim()),
            StringComparer.OrdinalIgnoreCase);

        foreach (var r in creates)
        {
            var code = r.AttributeCode.Trim();

            if (!seen.Add(code))
                return $"The attribute code '{code}' is already present in the database.";
        }

        return null;
    }

    private async Task Save()
    {
        var creates = rows.Where(newRows.Contains).ToList();
        var updates = rows.Where(IsModified).ToList();

        if (creates.Count + updates.Count == 0)
        {
            Toasts.Warning("Please make changes before saving.");
            return;
        }

        if (creates.Any(r => string.IsNullOrWhiteSpace(r.AttributeCode)))
        {
            Toasts.Warning("AttributeCode is required for new rows.");
            return;
        }

        var duplicate = FindDuplicate(creates);
        if (duplicate is not null)
        {
            dialogMessage = duplicate;
            return;
        }

        isSaving = true;

        try
        {
            foreach (var row in creates)
            {
                using var r = await Http.PostAsJsonAsync(Api, row);   // POST: new rows
                if (!await Check(r)) return;
            }

            foreach (var row in updates)
            {
                using var r = await Http.PutAsJsonAsync(KeyUrl(row), row);   // PUT: edited rows
                if (!await Check(r)) return;
            }

            Toasts.Success("Save successful");
            await Load();
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to save Retailer Attributes: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    // ---------- delete (dialogs, no toasts) ----------

    private void DeleteRow()
    {
        if (selectedRow is null)
            return;

        // unsaved new row: just drop it, nothing to delete on the server
        if (newRows.Contains(selectedRow))
        {
            newRows.Remove(selectedRow);
            rows.Remove(selectedRow);
            selectedRow = null;
            return;
        }

        confirmDeleteMessage = "Are you sure you want to delete the selected row?";
    }

    private async Task ConfirmDelete()
    {
        confirmDeleteMessage = null;

        if (selectedRow is null)
            return;

        isSaving = true;

        try
        {
            using var r = await Http.DeleteAsync(KeyUrl(selectedRow));

            if (r.IsSuccessStatusCode)
            {
                deleteResultMessage = "The row was deleted successfully.";
                await Load();
            }
            else
            {
                deleteResultMessage = await ReadMessage(r);
            }
        }
        catch (Exception ex)
        {
            deleteResultMessage = $"Unable to delete row: {ex.Message}";
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task<bool> Check(HttpResponseMessage response)
    {
        if (response.IsSuccessStatusCode)
            return true;

        Toasts.Error(await ReadMessage(response));
        return false;
    }
}

        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to save Retailer Attributes Services: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    // ---------- delete (dialogs, no toasts) ----------

    private void DeleteRow()
    {
        if (selectedRow is null)
            return;

        // unsaved new row: just drop it, nothing to delete on the server
        if (newRows.Contains(selectedRow))
        {
            newRows.Remove(selectedRow);
            rows.Remove(selectedRow);
            selectedRow = null;
            return;
        }

        confirmDeleteMessage = "Are you sure you want to delete the selected row?";
    }

    private async Task ConfirmDelete()
    {
        confirmDeleteMessage = null;

        if (selectedRow is null)
            return;

        isSaving = true;

        try
        {
            // delete the row as it is stored, even if its key was edited on screen
            var old = originalKeys[selectedRow];

            using var r = await Http.DeleteAsync(KeyUrl(old.Code, old.Service));

            if (r.IsSuccessStatusCode)
            {
                deleteResultMessage = "The row was deleted successfully.";
                await Load();
            }
            else
            {
                deleteResultMessage = await ReadMessage(r);
            }
        }
        catch (Exception ex)
        {
            deleteResultMessage = $"Unable to delete row: {ex.Message}";
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task<bool> Check(HttpResponseMessage response)
    {
        if (response.IsSuccessStatusCode)
            return true;

        Toasts.Error(await ReadMessage(response));
        return false;
    }
