private void HandleLoadedFiles()
{
    var allPaths = new[] { fichierProPath, fichierCommentPath, fichierQuiPath, fichierQuandPath, fichierPourquoiPath };

    if (allPaths.Any(t => !string.IsNullOrWhiteSpace(t) &&
        (t.StartsWith(@"C:\", StringComparison.OrdinalIgnoreCase) ||
         t.StartsWith(@"D:\", StringComparison.OrdinalIgnoreCase))))
    {
        CopyFiles();
    }

    options.Clear();
    programInfo.OndemandProgram!.Options = null!;
    // ... rest of your existing HandleLoadedFiles code
}


private void CopyFiles()
{
    if (!Directory.Exists(ShareDirectory))
        Directory.CreateDirectory(ShareDirectory);

    string subFolder = Path.Combine(ShareDirectory,
        Environment.UserName + "_" + DateTime.Now.ToString("yyyyMMddHHmmss"));
    Directory.CreateDirectory(subFolder);

    if (!string.IsNullOrWhiteSpace(fichierProPath) && File.Exists(fichierProPath))
    {
        var dest = Path.Combine(subFolder, Path.GetFileName(fichierProPath));
        File.Copy(fichierProPath, dest, true);
        fichierProPath = dest;
    }

    if (!string.IsNullOrWhiteSpace(fichierCommentPath) && File.Exists(fichierCommentPath))
    {
        var dest = Path.Combine(subFolder, Path.GetFileName(fichierCommentPath));
        File.Copy(fichierCommentPath, dest, true);
        fichierCommentPath = dest;
    }

    if (!string.IsNullOrWhiteSpace(fichierQuiPath) && File.Exists(fichierQuiPath))
    {
        var dest = Path.Combine(subFolder, Path.GetFileName(fichierQuiPath));
        File.Copy(fichierQuiPath, dest, true);
        fichierQuiPath = dest;
    }

    if (!string.IsNullOrWhiteSpace(fichierQuandPath) && File.Exists(fichierQuandPath))
    {
        var dest = Path.Combine(subFolder, Path.GetFileName(fichierQuandPath));
        File.Copy(fichierQuandPath, dest, true);
        fichierQuandPath = dest;
    }

    if (!string.IsNullOrWhiteSpace(fichierPourquoiPath) && File.Exists(fichierPourquoiPath))
    {
        var dest = Path.Combine(subFolder, Path.GetFileName(fichierPourquoiPath));
        File.Copy(fichierPourquoiPath, dest, true);
        fichierPourquoiPath = dest;
    }
}
