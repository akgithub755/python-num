    private readonly string ShareDirectory = Path.Combine(Constants.ROOT_ADHOC_PATH, @"Siceron\temp");

    private string fichierProPath = string.Empty;
    private string fichierCommentPath = string.Empty;
    private string fichierQuiPath = string.Empty;
    private string fichierQuandPath = string.Empty;
    private string fichierPourquoiPath = string.Empty;

    private async Task OpenFileDialog(string pickerId)
    {
        await JS.InvokeVoidAsync("fileHelpers.openFilePickerById", pickerId);
    }

    private void OnFileSelected(InputFileChangeEventArgs e, int index)
    {
        var file = e.File;
        if (file == null)
            return;

        var normalizedPath = Path.Combine(ShareDirectory, file.Name)
            .Replace('\\', Path.DirectorySeparatorChar)
            .Replace('/', Path.DirectorySeparatorChar);

        switch (index)
        {
            case 1: fichierProPath = normalizedPath; break;
            case 2: fichierCommentPath = normalizedPath; break;
            case 3: fichierQuiPath = normalizedPath; break;
            case 4: fichierQuandPath = normalizedPath; break;
            case 5: fichierPourquoiPath = normalizedPath; break;
        }
    }

        // Build parameter list and enqueue, matching your existing RequiredParameters pattern
    }

    private void ResetThisForm()
    {
        fichierProPath = fichierCommentPath = fichierQuiPath = fichierQuandPath = fichierPourquoiPath = string.Empty;
        nomClient = nomMarche = string.Empty;
        extrap = "False";
    }
