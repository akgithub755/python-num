import win32com.client as win32

def send_email():
    outlook = win32.Dispatch("Outlook.Application")
    mail = outlook.CreateItem(0)  # 0 = olMailItem

    mail.To = "friend1@example.com; friend2@example.com; friend3@example.com"
    # mail.CC = "someone@example.com"
    # mail.BCC = "hidden@example.com"

    mail.Subject = "Your Subject Here"
    mail.Body = "Hello,\n\nThis is the email body.\n\nRegards,\nYour Name"

    # To send immediately:
    mail.Send()

    # To just open a draft instead of sending automatically, use:
    # mail.Display()

if __name__ == "__main__":
    send_email()
