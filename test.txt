@code {
    private const string Api = "/api/RecapContourMapping";
    private const string ListUrl = "/api/RecapContourMapping/GetRecapContours";   // returns every row, no paging

    private record Col(
        string Title,
        Func<RecapContourMappingDto, string?> Get,
        Action<RecapContourMappingDto, string?>? Set = null,
        bool NewOnly = false,
        string InputType = "text");

    // Set == null -> read-only column, NewOnly -> editable only on newly added rows.
    // Mapping + Client are the key, so they are locked on saved rows.
    // The four user columns at the end stay editable on saved rows, like the original screen.
    private readonly List<Col> cols = new()
    {
        new("Mapping", x => x.Mapping,
            (x, v) => x.Mapping = v?.Trim() ?? string.Empty, true),
        new("Client", x => x.Client,
            (x, v) => x.Client = v?.Trim() ?? string.Empty, true),
        new("Libellé", x => x.Libelle,
            (x, v) => x.Libelle = v, true),
        new("Evol", x => x.Evol?.ToString(),
            (x, v) => x.Evol = int.TryParse(v, out var n) ? n : null, true, "number"),
        new("Date dernier lancement", x => x.DateDernierLancement?.ToString("yyyy-MM-ddTHH:mm"),
            (x, v) => x.DateDernierLancement = DateTime.TryParse(v, out var d) ? d : null, true, "datetime-local"),
        new("Check répart. evol", x => x.CheckRepartEvol,
            (x, v) => x.CheckRepartEvol = v, true),
        new("Check nouveau croisement histo", x => x.CheckNewCroisementHisto,
            (x, v) => x.CheckNewCroisementHisto = v, true),
        new("Check nouveau croisement EAN", x => x.CheckNewCroisementNewEan,
            (x, v) => x.CheckNewCroisementNewEan = v, true),
        new("Date validation user", x => x.DateValidationUser,
            (x, v) => x.DateValidationUser = v),
        new("Nom user", x => x.NomUser,
            (x, v) => x.NomUser = v),
        new("Commentaire user", x => x.CommentaireUser,
            (x, v) => x.CommentaireUser = v),
        new("Temps mapping", x => x.TempsMapping,
            (x, v) => x.TempsMapping = v),
    };

    private List<RecapContourMappingDto> rows = new();
    private Dictionary<RecapContourMappingDto, string> original = new();
    private readonly HashSet<RecapContourMappingDto> newRows = new();

    private RecapContourMappingDto? selectedRow;
    private RecapContourMappingDto? editingRow;
    private string? editingCol;

    private bool isLoading;
    private bool isSaving;
    private string? errorMessage;

    private string? dialogMessage;
    private string? confirmDeleteMessage;
    private string? deleteResultMessage;
    private string? confirmDiscardMessage;

    protected override async Task OnInitializedAsync() => await Load();

    // ---------- helpers ----------

    private static string DisplayValue(string? v) => string.IsNullOrWhiteSpace(v) ? "--" : v;

    private static string ShowValue(RecapContourMappingDto row, Col col)
    {
        var v = col.Get(row);

        if (col.InputType == "datetime-local")
            v = v?.Replace('T', ' ');

        return DisplayValue(v);
    }

    private bool CanEdit(RecapContourMappingDto row, Col col) =>
        col.Set is not null && (!col.NewOnly || newRows.Contains(row));

    private bool IsModified(RecapContourMappingDto row) =>
        !newRows.Contains(row) &&
        original.TryGetValue(row, out var json) &&
        json != JsonSerializer.Serialize(row);

    private string GetRowClass(RecapContourMappingDto row)
    {
        if (row == selectedRow) return "bg-blue-100";
        if (newRows.Contains(row)) return "bg-green-50";
        if (IsModified(row)) return "bg-yellow-50";
        return string.Empty;
    }

    private void BeginEdit(RecapContourMappingDto row, Col col)
    {
        selectedRow = row;

        if (!CanEdit(row, col))
            return;

        editingRow = row;
        editingCol = col.Title;
    }

    private void EndEdit()
    {
        editingRow = null;
        editingCol = null;
    }

    private void Change(RecapContourMappingDto row, Col col, string? value) =>
        col.Set!(row, string.IsNullOrWhiteSpace(value) ? null : value);

    // two-part key in the URL: /{mapping}/{client}
    private static string KeyUrl(RecapContourMappingDto row) =>
        $"{Api}/{Uri.EscapeDataString(row.Mapping)}/{Uri.EscapeDataString(row.Client)}";

    // shows only the message text, never the raw JSON
    private static async Task<string> ReadMessage(HttpResponseMessage r)
    {
        var body = await r.Content.ReadAsStringAsync();

        try
        {
            using var doc = JsonDocument.Parse(body);

            if (doc.RootElement.ValueKind == JsonValueKind.String)
                return doc.RootElement.GetString() ?? body;

            if (doc.RootElement.ValueKind == JsonValueKind.Object)
            {
                foreach (var p in doc.RootElement.EnumerateObject())
                    if (p.Name.Equals("message", StringComparison.OrdinalIgnoreCase) &&
                        p.Value.ValueKind == JsonValueKind.String)
                        return p.Value.GetString() ?? body;
            }
        }
        catch (Exception)
        {
            // body was plain text, show it as-is
        }

        return string.IsNullOrWhiteSpace(body)
            ? $"Request failed. HTTP {(int)r.StatusCode}"
            : body;
    }

    // ---------- load / refresh ----------

    private async Task Load()
    {
        isLoading = true;
        errorMessage = null;

        try
        {
            rows = await Http.GetFromJsonAsync<List<RecapContourMappingDto>>(ListUrl) ?? new();

            original = rows.ToDictionary(r => r, r => JsonSerializer.Serialize(r));
            newRows.Clear();
            selectedRow = null;
            EndEdit();
        }
        catch (HttpRequestException ex)
        {
            errorMessage = $"Unable to load Recap Contour data: {ex.Message}";
        }
        catch (Exception ex)
        {
            errorMessage = $"An unexpected error occurred: {ex.Message}";
        }
        finally
        {
            isLoading = false;
        }
    }

    private async Task Refresh()
    {
        if (newRows.Count > 0 || rows.Any(IsModified))
        {
            confirmDiscardMessage = "You have unsaved changes. Do you want to discard them and refresh?";
            return;
        }

        await DoRefresh();
    }

    private async Task ConfirmDiscard()
    {
        confirmDiscardMessage = null;
        await DoRefresh();
    }

    private async Task DoRefresh()
    {
        await Load();

        if (errorMessage is null)
            Toasts.Success("Recap Contour refreshed successfully.");
    }

    // ---------- add / cancel / save ----------

    private void AddRow()
    {
        var row = new RecapContourMappingDto
        {
            Mapping = string.Empty,
            Client = string.Empty
        };

        newRows.Add(row);
        rows.Insert(0, row);
        selectedRow = row;
    }

    private void CancelAdd()
    {
        rows.RemoveAll(newRows.Contains);   // drops the unsaved new row(s) from the grid
        newRows.Clear();
        selectedRow = null;
        EndEdit();
    }

    // the key is the combination of Mapping and Client
    private string? FindDuplicate(List<RecapContourMappingDto> creates)
    {
        var seen = rows.Where(x => !newRows.Contains(x))
                       .Select(x => (x.Mapping.Trim().ToLowerInvariant(), x.Client.Trim().ToLowerInvariant()))
                       .ToHashSet();

        foreach (var r in creates)
        {
            var mapping = r.Mapping.Trim();
            var client = r.Client.Trim();

            if (!seen.Add((mapping.ToLowerInvariant(), client.ToLowerInvariant())))
                return $"The combination Mapping '{mapping}' and Client '{client}' " +
                       "is already present in the database.";
        }

        return null;
    }

    private async Task Save()
    {
        var creates = rows.Where(newRows.Contains).ToList();
        var updates = rows.Where(IsModified).ToList();

        if (creates.Count + updates.Count == 0)
        {
            Toasts.Warning("Please make changes before saving.");
            return;
        }

        if (creates.Any(r => string.IsNullOrWhiteSpace(r.Mapping) || string.IsNullOrWhiteSpace(r.Client)))
        {
            Toasts.Warning("Mapping and Client are required for new rows.");
            return;
        }

        var duplicate = FindDuplicate(creates);
        if (duplicate is not null)
        {
            dialogMessage = duplicate;
            return;
        }

        isSaving = true;

        try
        {
            foreach (var row in creates)
            {
                using var r = await Http.PostAsJsonAsync(Api, row);   // POST: new rows
                if (!await Check(r)) return;
            }

            foreach (var row in updates)
            {
                using var r = await Http.PutAsJsonAsync(KeyUrl(row), row);   // PUT: edited rows
                if (!await Check(r)) return;
            }

            Toasts.Success("Save successful");
            await Load();
        }
        catch (HttpRequestException ex)
        {
            Toasts.Error($"Unable to save Recap Contour: {ex.Message}");
        }
        catch (Exception ex)
        {
            Toasts.Error($"An unexpected error occurred: {ex.Message}");
        }
        finally
        {
            isSaving = false;
        }
    }

    // ---------- delete (dialogs, no toasts) ----------

    private void DeleteRow()
    {
        if (selectedRow is null)
            return;

        // unsaved new row: just drop it, nothing to delete on the server
        if (newRows.Contains(selectedRow))
        {
            newRows.Remove(selectedRow);
            rows.Remove(selectedRow);
            selectedRow = null;
            return;
        }

        confirmDeleteMessage = "Are you sure you want to delete the selected row?";
    }

    private async Task ConfirmDelete()
    {
        confirmDeleteMessage = null;

        if (selectedRow is null)
            return;

        isSaving = true;

        try
        {
            using var r = await Http.DeleteAsync(KeyUrl(selectedRow));

            if (r.IsSuccessStatusCode)
            {
                deleteResultMessage = "The row was deleted successfully.";
                await Load();
            }
            else
            {
                deleteResultMessage = await ReadMessage(r);
            }
        }
        catch (Exception ex)
        {
            deleteResultMessage = $"Unable to delete row: {ex.Message}";
        }
        finally
        {
            isSaving = false;
        }
    }

    private async Task<bool> Check(HttpResponseMessage response)
    {
        if (response.IsSuccessStatusCode)
            return true;

        Toasts.Error(await ReadMessage(response));
        return false;
    }
}
