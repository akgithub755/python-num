private void HandleLoadedFiles()
{
    var allPaths = new[] { fichierProPath, fichierCommentPath, fichierQuiPath, fichierQuandPath, fichierPourquoiPath };

    if (allPaths.Any(t => !string.IsNullOrWhiteSpace(t) &&
        (t.StartsWith(@"C:\", StringComparison.OrdinalIgnoreCase) ||
         t.StartsWith(@"D:\", StringComparison.OrdinalIgnoreCase))))
    {
        CopyFiles();
    }

    options.Clear();
    programInfo.OndemandProgram!.Options = null!;
    programInfo.OndemandProgram.Comments = string.Empty;
    comments.Clear();
    programInfo.UserComments = string.Empty;

#if DEBUG
    options.Insert(0, "-O");
#endif

    options.Add(Constants.ROOT_ADHOC_PATH + @"\Siceron\pgm\SiceronKFFFormatter.py");

    if (!string.IsNullOrEmpty(fichierProPath)) { options.Add("pro=" + fichierProPath); } else { options.Add(fichierProPath); }
    if (!string.IsNullOrEmpty(fichierCommentPath)) { options.Add("cmt=" + fichierCommentPath); } else { options.Add(fichierCommentPath); }
    if (!string.IsNullOrEmpty(fichierQuiPath)) { options.Add("qui=" + fichierQuiPath); } else { options.Add(fichierQuiPath); }
    if (!string.IsNullOrEmpty(fichierQuandPath)) { options.Add("qnd=" + fichierQuandPath); } else { options.Add(fichierQuandPath); }
    if (!string.IsNullOrEmpty(fichierPourquoiPath)) { options.Add("prq=" + fichierPourquoiPath); } else { options.Add(fichierPourquoiPath); }
    if (!string.IsNullOrEmpty(nomClient)) { options.Add("cli=" + nomClient); } else { options.Add(nomClient); }
    if (!string.IsNullOrEmpty(nomMarche)) { options.Add("mrkt=" + nomMarche); } else { options.Add(nomMarche); }

    options.Add("extrap=" + extrap);

    options = programInfo.OndemandProgram!.FormatParameters(options);
    programInfo.OndemandProgram.Options = options;
    programInfo.OndemandProgram.requiredParameters = RequiredParameters;
    programInfo.OndemandProgram.optionalParameters = OptionalParameters;

    comments.Append($"{programInfo.OndemandProgram.Programname}" +
                     $"\n\tfichierProPath : {fichierProPath}" +
                     $"\n\tnomClient : {nomClient}" +
                     $"\n\tnomMarche : {nomMarche}" +
                     $"\n\textrap : {extrap}" +
                     $"\n\tfichierCommentPath : {fichierCommentPath}" +
                     $"\n\tfichierQuiPath : {fichierQuiPath}" +
                     $"\n\tfichierQuandPath : {fichierQuandPath}" +
                     $"\n\tfichierPourquoiPath : {fichierPourquoiPath}"
    );

    programInfo.OndemandProgram.Comments = comments.ToString().Trim();
    programInfo.UserComments = programInfo.OndemandProgram.Comments;
}
